// POPRAWKA 1: Dynamiczny środek parcia aero (CoP) zależny od prześwitu przód/tył
// Zastępuje statycznerideHeightLift oraz sztywny podział 0.46 w updatePhysics

/**
 * Zwraca mnożnik docisku ORAZ dynamiczny udział przedniej osi (CoP)
 * na podstawie chwilowego prześwitu przedniego (frh) i tylnego (rrh).
 */
static rideHeightAero(frh: number, rrh: number, stall?: number): { lift: number; frontShare: number } {
  // Sprawność przedniej części podłogi / skrzydła (optimum ok. 16 mm)
  const frontEff = 1 + 0.35 * Math.exp(-(((frh - 0.016) / 0.018) ** 2));
  // Sprawność dyfuzora tylnego (optimum ok. 35 mm)
  const rearEff = 1 + 0.45 * (rrh / 0.035) * Math.exp(1 - rrh / 0.035);
  
  // Zerwanie strugi przy zbyt niskim prześwicie
  stall ??= frh < 0.008 || rrh < 0.015
    ? 0.75 + 0.25 * Math.max(0, Math.min(1, Math.min(frh / 0.008, rrh / 0.015)))
    : 1;

  // Całkowity mnożnik docisku znormalizowany do 1.0 w punkcie referencyjnym
  const lift = (frontEff * rearEff * stall) / 1.83;

  // Dynamiczny balans aero (CoP): bazowo 0.46, przesuwa się do przodu podczas nurkowania (niski przód)
  const balanceShift = ((frontEff / 1.35) - (rearEff / 1.45)) * 0.08;
  const frontShare = Math.max(0.38, Math.min(0.54, 0.46 + balanceShift));

  return { lift, frontShare };
}

// POPRAWKA 4: Dwuwarstwowy model termiczny opony (Powierzchnia + Rdzeń)
// Zastępuje metodę stepTyre oraz tyreGrip w Car.ts
// ============================================================================

// Dodatkowe pola w klasie Car (temperatura powierzchni i rdzenia):
public tyreSurfaceTempFront = 85;
public tyreSurfaceTempRear = 85;
public static readonly TYRE_SURFACE_CAPACITY = 1600; // J/K (szybka reakcja bieżnika: 2-3 s)
public static readonly TYRE_CORE_CAPACITY = 10400;   // J/K (wolna reakcja karkasu: ~40 s)
public static readonly TYRE_INTERNAL_COND = 220;     // W/K (przewodzenie powierzchnia <-> rdzeń)

private stepTyre(front: boolean, slipPower: number, load: number, speed: number, dt: number): void {
  let tSurf = front ? this.tyreSurfaceTempFront : this.tyreSurfaceTempRear;
  let tCore = front ? this.tyreTempFront : this.tyreTempRear;
  const wear = front ? this.tyreWearFront : this.tyreWearRear;
  const wet = this.wetness;
  const air = Car.config.weather.airTemp;
  const trackTemp = Car.config.weather.trackTemp;

  // Ciepło z poślizgu grzeje powierzchnię; histereza od ugięcia grzeje bezpośrednio rdzeń
  const surfaceHeat = slipPower * Car.TYRE_HEAT_SCALE;
  const coreHeat = (0.0075 * load * speed) * Car.TYRE_HEAT_SCALE;

  // Chłodzenie powierzchni powietrzem, kontaktem z asfaltem i wodą
  const externalCooling = (2 * (15 + 3.2 * Math.pow(Math.max(0, speed), 0.8)) * (tSurf - air)
    + 60 * (tSurf - trackTemp)) * (1 + 0.9 * wet);

  // Wymiana ciepła między powierzchnią a rdzeniem opony
  const internalTransfer = Car.TYRE_INTERNAL_COND * (tSurf - tCore);

  // Całkowanie temperatur powierzchni i rdzenia
  tSurf = Math.max(air - 8 * wet, Math.min(200,
    tSurf + (surfaceHeat - externalCooling - internalTransfer) / Car.TYRE_SURFACE_CAPACITY * dt));
  tCore = Math.max(air - 8 * wet, Math.min(160,
    tCore + (coreHeat + internalTransfer) / Car.TYRE_CORE_CAPACITY * dt));

  // Efektywna temperatura dla przyczepności (70% powierzchnia, 30% rdzeń)
  const effectiveTemp = 0.7 * tSurf + 0.3 * tCore;

  // Zużycie zależy silnie od przegrzania powierzchni
  const wearRate = Car.TYRE_WEAR_RATE * COMPOUND_SPECS[this.tyreCompound].wear * (1 - 0.6 * wet);
  const nextWear = Math.min(1, wear + wearRate * Math.pow(effectiveTemp / 100, 2) * slipPower * dt);

  if (front) {
    this.tyreSurfaceTempFront = tSurf;
    this.tyreTempFront = tCore;
    this.tyreWearFront = nextWear;
  } else {
    this.tyreSurfaceTempRear = tSurf;
    this.tyreTempRear = tCore;
    this.tyreWearRear = nextWear;
  }
}
